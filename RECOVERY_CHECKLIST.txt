# Verificar que Odoo tiene rates:
sudo docker exec bridge-amazon-inbound-worker python3 -c "
import requests, sqlite3
db = sqlite3.connect('/data/bridge.db')
def gs(k): return db.execute('SELECT value FROM bridge_settings WHERE key=?',(k,)).fetchone()[0]
url,db_,user,pw = gs('odoo_url'),gs('odoo_db'),gs('odoo_user'),gs('odoo_password')
uid = requests.post(url+'/jsonrpc',json={'jsonrpc':'2.0','method':'call','id':1,'params':{'service':'common','method':'authenticate','args':[db_,user,pw,{}]}},timeout=10).json()['result']
rates = requests.post(url+'/jsonrpc',json={'jsonrpc':'2.0','method':'call','id':2,'params':{'service':'object','method':'execute_kw','args':[db_,uid,pw,'res.currency.rate','search_read',[[['currency_id.name','=','USD']]],{'fields':['rate','name'],'order':'name desc','limit':3}]}},timeout=10).json()['result']
for r in (rates or []): print(r['name'], round(1.0/r['rate'],4),'MXN/USD')
"
